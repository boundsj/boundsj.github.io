{{- /* Page context by default. As a dict: page, hideCaption, and optional link/kind to make the image a (non-focusable, duplicate) link to the post. Animated GIF heroes get a still (featuredImageStill, or the GIF's first frame rendered as PNG) that shows under reduced motion, plus a pause/play button wired up by site.js. */ -}} {{- $page := . -}} {{- $hideCaption := false -}} {{- $link := "" -}} {{- $kind := "" -}} {{- if reflect.IsMap . -}} {{- $page = .page -}} {{- $hideCaption = .hideCaption | default false -}} {{- $link = .link | default "" -}} {{- $kind = .kind | default "" -}} {{- end -}} {{- with $page.Resources.GetMatch "featured-image" -}} {{- $img := . -}} {{- $animated := eq $img.MediaType.SubType "gif" -}} {{- $still := false -}} {{- with $page.Params.featuredImageStill }}{{ $still = $page.Resources.GetMatch . }}{{ end -}} {{- if and $animated (not $still) }}{{ $still = $img.Resize (printf "%dx png" (int (math.Min 1280 $img.Width))) }}{{ end -}} {{- $id := path.Base $page.RelPermalink -}} {{- end -}}
