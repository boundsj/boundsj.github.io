{{- /* A compact inline figure that opens a detailed version of itself. {{< zoom-figure src="summary.svg" full="detail.svg" fullMobile="detail-mobile.svg" alt="Summary diagram: …" fullAlt="…" label="Before: me as the router" caption="…" >}} src the inline image, sized for the prose column full the detailed image; also the plain link target without JavaScript fullMobile optional detailed variant for screens up to 600px wide alt describes the inline image; fullAlt describes the detailed one label names the figure in the viewer's title bar caption optional, markdown site.js (figure viewer) opens `full` in a dialog that fits it to the screen, with an actual-size toggle for scrolling. Bundle files only; SVG sizes come from their viewBox. Every URL is fingerprinted (content hash in the file name, as the site does for CSS and JS), so a revised image always gets a new URL and no browser keeps showing a cached old one. */ -}} {{- $p := .Page -}} {{- $src := $p.Resources.GetMatch (.Get "src") -}} {{- $full := $p.Resources.GetMatch (.Get "full") -}} {{- $mobile := false -}} {{- with .Get "fullMobile" }}{{ $mobile = $p.Resources.GetMatch . }}{{ end -}} {{- with $src }}{{ $src = . | fingerprint }}{{ end -}} {{- with $full }}{{ $full = . | fingerprint }}{{ end -}} {{- with $mobile }}{{ $mobile = . | fingerprint }}{{ end -}} {{- if not (and $src $full) -}} {{- warnf "[zoom-figure] %s: src %q or full %q is not in the bundle" $p.File.Path (.Get "src") (.Get "full") -}} {{- else -}} {{- $s := partial "zoom-figure-size.html" $src -}} {{- $f := partial "zoom-figure-size.html" $full -}} {{- end -}}
